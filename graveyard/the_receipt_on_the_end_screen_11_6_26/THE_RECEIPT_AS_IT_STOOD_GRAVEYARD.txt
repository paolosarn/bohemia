==== THE CODE (tillStart, tillLine, tillSync, tillTear) ====
/* ---------- __THE_RECEIPT__ (RUN FIVE, rule 405(c)): the fight's receipt, printed into the one end screen ---------- */
var TILL = {q:[], t:null, got:[], cash:false, total:0, n:0}, TILL_BEAT = 125, TILL_TEAR_MS = 320, TILL_W = 20;   /* a line every quarter beat at 120 BPM; 20 characters a line */
function pad(v, w, right){ v = String(v); if(v.length > w) v = v.slice(0, w); var sp = new Array(w - v.length + 1).join(' '); return right ? sp + v : v + sp; }
function two(l, r){ l = String(l); r = String(r); var n = Math.max(1, TILL_W - l.length - r.length); return (l + new Array(n + 1).join(' ') + r).slice(0, Math.max(TILL_W, l.length + r.length + 1)); }
function tillLine(text, cls, data){ TILL.q.push({text:text, cls:cls || '', data:data || null}); if(!TILL.t) TILL.t = setTimeout(tillTick, 0); }
function tillTick(){
  TILL.t = null; var L = TILL.q.shift(); if(!L) return;
  var el = document.getElementById('tillpaper'); if(!el) return;
  var d = document.createElement('div'); d.className = 'tl new' + (L.cls ? ' ' + L.cls : ''); d.textContent = L.text;
  if(L.data) for(var k in L.data) d.dataset[k] = L.data[k];
  if(L.id){ d.id = L.id; } el.appendChild(d); TILL.n++; if(!(window.scrollY > 0)) tillTop();
  setTimeout(function(){ d.classList.remove('new'); }, TILL_BEAT + 20);
  if(TILL.q.length) TILL.t = setTimeout(tillTick, TILL_BEAT);
}
function tillFlush(){ while(TILL.q.length) tillTick(); if(TILL.t){ clearTimeout(TILL.t); TILL.t = null; } }   /* the tear prints what is left at once */
function tillTop(){ var b = document.getElementById('bar'), t = document.getElementById('tillclip'); if(!b || !t) return;   /* from under the bar's foot, wherever the bar stands (in the game the map's own bar is over it) */
  var y = Math.round(b.getBoundingClientRect().bottom + (window.scrollY || 0) - 8); if(t.dataset.y !== String(y)){ t.dataset.y = y; t.style.setProperty('--till-top', Math.max(0, y) + 'px'); } }
function tillStart(f, rc, head){
  var el = document.getElementById('tillpaper'), t = document.getElementById('till'); if(!el || !t) return;
  if(TILL.t) clearTimeout(TILL.t); TILL = {q:[], t:null, got:[], cash:false, total:0, n:0};
  el.textContent = ''; t.className = ''; tillTop();
  var rule = new Array(TILL_W + 1).join('-');
  tillLine(two('BOHEMIA', 'NO.' + ('0000' + Math.floor(Math.abs(+S.seq || 0)) % 10000).slice(-4)));   /* draft:true */
  if(head) tillLine(head, 'big');
  var tally = [];
  if(f.rounds) tally.push(two(f.rounds + (f.rounds === 1 ? ' ROUND' : ' ROUNDS'), rc.secs ? mmss(rc.secs) : ''));
  if(rc.theirs) tally.push(two('DOWN ' + (rc.down | 0) + ' OF ' + rc.theirs, rc.ran ? 'RAN ' + rc.ran : ''));
  /* #tally and #rounds keep their names: the fight's numbers, now on the receipt */
  if(tally.length){ var tq = {text:tally.join('\n'), cls:'', data:null}; tq.id = 'tally'; TILL.q.push(tq); if(!TILL.t) TILL.t = setTimeout(tillTick, 0); }
  (rc.lines || []).forEach(function(x){ tillLine(x); });
  var men = (rc.men || []);
  if(men.length){
    tillLine(rule, 'rule');
    tillLine(pad('WHO', 6) + pad('K', 2, true) + pad('BLD', 4, true) + pad('TKN', 4, true) + pad('XP', 4, true));
    men.forEach(function(n){ var m = S.men.filter(function(x){ return x.name === n.name; })[0] || {};
      tillLine(pad(m.main ? 'YOU' : n.name, 6) + pad(n.kills | 0, 2, true) + pad(n.dealt | 0, 4, true) + pad(n.taken | 0, 4, true) + pad(m.gained | 0, 4, true), '',
        {man:m.main ? 'YOU' : n.name, kills:n.kills | 0, dealt:n.dealt | 0, taken:n.taken | 0}); });
  }
  tillLine(rule, 'rule');
  if(!S.won) tillLine(f.arena ? 'THE PURSE IS PAID AT THE ROPE' : 'NOTHING TAKEN');   /* draft:true */
  t.onclick = function(){ if(!t.classList.contains('torn')) t.classList.toggle('open'); };
  window.__TILL = TILL;
}
/* the loot counted onto it: every piece in the stash that came off them prints once with what a town pays for it; a piece put back
   prints VOID; the counts print when they are taken */
function tillSync(){
  if(!S.fight || !document.getElementById('tillpaper')) return;
  var now = S.stash.filter(function(it){ return it && it.looted && it.fresh; });
  now.forEach(function(it){ if(TILL.got.indexOf(it) >= 0) return; TILL.got.push(it); var v = SP.sellBack(it, S.at) | 0; TILL.total += v;
    tillLine(two('+ ' + shortName(it).toUpperCase(), String(v)), '', {item:it.id}); });
  TILL.got.slice().forEach(function(it){ if(now.indexOf(it) >= 0) return; TILL.got.splice(TILL.got.indexOf(it), 1); var v = SP.sellBack(it, S.at) | 0; TILL.total -= v;
    tillLine(two('VOID ' + shortName(it).toUpperCase(), '-' + v), 'void', {item:it.id}); });
  if(S.countsTaken && !TILL.cash){ TILL.cash = true; var c = S.counts || {};
    if(c.batteries > 0){ TILL.total += c.batteries | 0; tillLine(two('+' + c.batteries + (c.batteries === 1 ? ' BATTERY' : ' BATTERIES'), String(c.batteries))); }
    if(c.rounds > 0) tillLine('+' + c.rounds + (c.rounds === 1 ? ' ROUND' : ' ROUNDS'));
    if(c.tape > 0) tillLine('+' + c.tape + ' TAPE'); }
}
/* LEAVE tears it off: the total prints, then the roll goes; false when there is nothing to tear (no receipt on the page) */
function tillTear(){
  var t = document.getElementById('till'); if(!t || !S.fight) return false;
  tillSync();
  if(S.won){ tillLine(new Array(TILL_W + 1).join('='), 'rule'); tillLine(two('TOTAL', String(TILL.total)), 'big'); }
  tillFlush(); t.classList.remove('open');
  void t.offsetWidth; t.classList.add('torn');
  return true;
}
window.addEventListener('resize', tillTop);
